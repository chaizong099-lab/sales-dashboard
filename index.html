    async function loadData() {

      applyRoleUI();

      if (!db) return;


      const { data, error } = await db

        .from("sales_projects")

        .select("*")

        .order("amount", { ascending: false });


      if (error) {

        document.getElementById("lastUpdatedText").textContent = "读取云端失败：" + error.message;

        return;

      }


      projects = (data || []).map(row => ({

        timeline: Array.isArray(row.timeline) ? row.timeline : []

      }));


      const now = new Date();

      document.getElementById("lastUpdatedText").textContent =

        `云端实时同步中 | 共 ${projects.length} 个项目 | 最近同步：${now.toLocaleTimeString("zh-CN")}`;


      renderAll();

    }


    function stageToStatus(stage) {

    function getFilteredProjects() {

      const kw = document.getElementById("searchInput").value.trim().toLowerCase();

      const owner = document.getElementById("ownerFilter").value;

      const status = document.getElementById("statusFilter").value;

      const sort = document.getElementById("sortFilter").value;


      let list = projects.filter(p => {

        if (owner !== "ALL" && p.owner !== owner) return false;

        if (status !== "ALL" && p.status !== status) return false;

        if (kw) {

          const hay = `${p.client} ${p.location} ${p.contact} ${p.scope} ${p.category} ${p.summary}`.toLowerCase();

          if (!hay.includes(kw)) return false;

        }

        return true;

      });


      list.sort((a, b) => {

        if (sort === "amountDesc") return (b.amount || 0) - (a.amount || 0);

        if (sort === "dateDesc") return (b.startDate || "").localeCompare(a.startDate || "");

        if (sort === "dateAsc") return (a.startDate || "").localeCompare(b.startDate || "");

        return 0;

      });

      return list;

    }


    function updateKPIs() {

      const active = projects.filter(p => p.status === "跟进中");

      const hot = projects.filter(p => p.stage === "方案报价中" || p.stage === "招投标/商务谈判");

      const won = projects.filter(p => p.status === "已成交");

      const lost = projects.filter(p => p.status === "已丢单");


      const sum = arr => arr.reduce((acc, cur) => acc + (Number(cur.amount) || 0), 0);


      document.getElementById("kpiActiveAmount").textContent = `¥${sum(active).toFixed(2)} 万`;

      document.getElementById("kpiActiveCount").textContent = `${active.length} 个项目`;


      document.getElementById("kpiHotAmount").textContent = `¥${sum(hot).toFixed(2)} 万`;

      document.getElementById("kpiHotCount").textContent = `${hot.length} 个项目`;


      document.getElementById("kpiWonAmount").textContent = `¥${sum(won).toFixed(2)} 万`;

      document.getElementById("kpiWonCount").textContent = `${won.length} 个项目`;


      const closedCount = won.length + lost.length;

      const winRate = closedCount > 0 ? ((won.length / closedCount) * 100).toFixed(1) : 0;

      document.getElementById("kpiWinRate").textContent = `已闭环项目胜率：${winRate}% (${won.length}/${closedCount})`;


      document.getElementById("kpiLostAmount").textContent = `¥${sum(lost).toFixed(2)} 万`;

      document.getElementById("kpiLostCount").textContent = `${lost.length} 个项目`;

    }

    async function saveProject(e) {

      e.preventDefault();

      if (!IS_SALES_EDITABLE || !db) return;


      const id = document.getElementById("editId").value;

      const stage = document.getElementById("fStage").value;

      const newLog = document.getElementById("fNewLog").value.trim();

      let summary = document.getElementById("fSummary").value.trim();


      const todayStr = new Date().toLocaleDateString("zh-CN", { month: "2-digit", day: "2-digit" }).replace("/", "");

      if (newLog) {

        summary = `${todayStr} ${newLog}`;

      }


      const existing = id ? projects.find(p => p.id === id) : null;

      const timeline = existing ? [...(existing.timeline || [])] : [];

      if (newLog) {

        timeline.push(`【${todayStr} 实时更新】${newLog}`);

      } else if (!existing) {

        timeline.push(`【${todayStr} 创建】新建商机录入`);

      }


      const rowPayload = {

        id: id || ("NEW-" + Date.now()),

        owner: document.getElementById("fOwner").value,

        start_date: document.getElementById("fStartDate").value || "2026-11-24",

        client: document.getElementById("fClient").value.trim(),

        location: document.getElementById("fLocation").value.trim() || "未标注",

        contact: document.getElementById("fContact").value.trim() || "未填写",

        phone: document.getElementById("fPhone").value.trim() || "未填写",

        scope: document.getElementById("fScope").value.trim() || "待明确",

        amount: parseFloat(document.getElementById("fAmount").value) || 0,

        status: stageToStatus(stage),

        stage: stage,

        category: document.getElementById("fCategory").value.trim() || "综合自动化",

        summary: summary || "正常推进中",

        timeline: timeline,

        updated_at: new Date().toISOString()

      };


      const { error } = await db.from("sales_projects").upsert([rowPayload]);

      if (error) {

        alert("保存失败：" + error.message);

        return;

      }


      closeModal();

      await loadData();

    }

    function exportCSV() {

      const headers = ["项目编号", "负责人", "启动时间", "客户名称", "项目地址", "对接人", "联系方式", "项目体量", "预计成交额(万元)", "当前状态", "当前阶段", "最新进展摘要"];

        `"${(p.summary || "").replace(/"/g, '""')}"`

      ]);

      const csvContent = "\uFEFF" + [headers.join(","), ...rows.map(r => r.join(","))].join("\n");

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });

      const url = URL.createObjectURL(blob);

      a.href = url;

      a.download = `公司销售资源进度总表_${new Date().toISOString().split("T")[0]}.csv`;

      a.click();

    }
